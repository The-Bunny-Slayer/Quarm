get_full_obtainment_details(p_item_id INT)

This database function takes an item ID and returns a JSON object representing the full, recursive crafting tree or obtainment information (vendor or drop) for that item.

SQL Definition


CREATE OR REPLACE FUNCTION get_full_obtainment_details(p_item_id integer)
 RETURNS jsonb
 LANGUAGE plpgsql
AS $function$
DECLARE
    result jsonb;
    recipe_info RECORD;
    component_info RECORD;
    final_components jsonb[] := '{}';
    obtainment_sources jsonb;
BEGIN
    -- First, check if the item is crafted.
    SELECT INTO recipe_info *
    FROM tradeskill_recipe r
    JOIN tradeskill_recipe_entries tre ON r.id = tre.recipe_id
    WHERE tre.item_id = p_item_id AND tre.successcount > 0
    LIMIT 1;

    -- If it is a crafted item, process its components.
    IF FOUND THEN
        FOR component_info IN
            SELECT i.id as item_id, tre.componentcount as qty, i.name as item_name, i.icon
            FROM tradeskill_recipe_entries tre
            JOIN items i ON i.id = tre.item_id
            WHERE tre.recipe_id = recipe_info.recipe_id AND tre.componentcount > 0
        LOOP
            -- For each component, recursively get its details.
            final_components := array_append(final_components, jsonb_build_object(
                'item_data', jsonb_build_object('id', component_info.item_id, 'name', component_info.item_name, 'icon', component_info.icon, 'qty', component_info.qty),
                'obtainment', get_full_obtainment_details(component_info.item_id)
            ));
        END LOOP;

        result := jsonb_build_object(
            'type', CASE WHEN recipe_info.tradeskill = 100 THEN 'quest' ELSE 'craft' END,
            'tradeskill', (SELECT name FROM tradeskill_enum WHERE id = recipe_info.tradeskill),
            'trivial', recipe_info.trivial,
            'notes', recipe_info.notes, -- Quest NPC name stored here
            'components', final_components
        );
        RETURN result;
    END IF;

    -- If not crafted, find vendors and drop sources.
    SELECT jsonb_agg(source_details ORDER BY type) INTO obtainment_sources FROM (
        -- VENDORS
        SELECT
            'vendor' AS type,
            nt.name AS name,
            z.long_name AS zone
        FROM merchantlist ml
        JOIN npc_types nt ON nt.merchant_id = ml.merchantid
        JOIN spawnentry se ON se.npcID = nt.id
        JOIN spawn2 s2 ON s2.spawngroupID = se.spawngroupID
        JOIN zone z ON z.short_name = s2.zone
        WHERE ml.item = p_item_id AND nt.class = 41
        GROUP BY nt.name, z.long_name

        UNION ALL

        -- MOB DROPS
        SELECT
            'mob_drop' AS type,
            REPLACE(nt.name, '_', ' ') AS name,
            z.long_name AS zone
        FROM loottable_entries lte
        JOIN lootdrop_entries lde ON lde.lootdrop_id = lte.lootdrop_id
        JOIN npc_types nt ON nt.loottable_id = lte.loottable_id
        JOIN spawnentry se ON se.npcID = nt.id
        JOIN spawn2 s2 ON s2.spawngroupID = se.spawngroupID
        JOIN zone z ON z.short_name = s2.zone
        WHERE lde.item_id = p_item_id
        GROUP BY nt.name, z.long_name
    ) AS source_details;

    result := jsonb_build_object(
        'type', 'obtainment',
        'sources', obtainment_sources
    );

    RETURN result;
END;
$function$;
