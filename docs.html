get_full_obtainment_details(p_item_id INT, p_visited INT[] DEFAULT '{}')

This database function takes an item ID and returns a JSON object representing the full, recursive crafting tree or obtainment information. It includes cycle detection to prevent infinite loops in recipes.

SQL Definition


CREATE OR REPLACE FUNCTION get_full_obtainment_details(p_item_id integer, p_visited integer[] DEFAULT '{}')
 RETURNS jsonb
 LANGUAGE plpgsql
AS $function$
DECLARE
    result jsonb;
    recipe_info RECORD;
    component_info RECORD;
    final_components jsonb[] := '{}';
    obtainment_sources jsonb;
BEGIN
    -- Cycle detection
    IF p_item_id = ANY(p_visited) THEN
        RETURN jsonb_build_object('type', 'error', 'message', 'Recursive loop detected');
    END IF;

    -- Add current item to visited list
    p_visited := array_append(p_visited, p_item_id);

    -- Check if the item is crafted
    SELECT INTO recipe_info *
    FROM tradeskill_recipe r
    WHERE r.id IN (SELECT recipe_id FROM tradeskill_recipe_entries WHERE item_id = p_item_id AND successcount > 0)
    LIMIT 1;

    IF FOUND THEN
        FOR component_info IN
            SELECT i.id as item_id, tre.componentcount as qty, i.name as item_name, i.icon
            FROM tradeskill_recipe_entries tre
            JOIN items i ON i.id = tre.item_id
            WHERE tre.recipe_id = recipe_info.id AND tre.componentcount > 0
        LOOP
            final_components := array_append(final_components, jsonb_build_object(
                'item_data', jsonb_build_object('id', component_info.item_id, 'name', component_info.item_name, 'icon', component_info.icon, 'qty', component_info.qty),
                'obtainment', get_full_obtainment_details(component_info.item_id, p_visited)
            ));
        END LOOP;

        result := jsonb_build_object(
            'type', CASE WHEN recipe_info.tradeskill = 100 THEN 'quest' ELSE 'craft' END,
            'tradeskill', CASE recipe_info.tradeskill
                WHEN 59 THEN 'Alchemy'
                WHEN 60 THEN 'Baking'
                WHEN 63 THEN 'Blacksmithing'
                WHEN 65 THEN 'Brewing'
                WHEN 64 THEN 'Fletching'
                WHEN 68 THEN 'Jewelry Making'
                WHEN 56 THEN 'Make Poison'
                WHEN 69 THEN 'Pottery'
                WHEN 58 THEN 'Research'
                WHEN 61 THEN 'Tailoring'
                WHEN 57 THEN 'Tinkering'
                ELSE 'Unknown'
            END,
            'trivial', recipe_info.trivial,
            'notes', recipe_info.notes,
            'components', final_components
        );
        RETURN result;
    END IF;

    -- If not crafted, find vendors and drop sources
    SELECT jsonb_agg(source_details ORDER BY type) INTO obtainment_sources FROM (
        SELECT 'vendor' AS type, nt.name, z.long_name AS zone
        FROM merchantlist ml
        JOIN npc_types nt ON nt.merchant_id = ml.merchantid
        JOIN spawnentry se ON se.npcID = nt.id
        JOIN spawn2 s2 ON s2.spawngroupID = se.spawngroupID
        JOIN zone z ON z.short_name = s2.zone
        WHERE ml.item = p_item_id AND nt.class = 41
        GROUP BY nt.name, z.long_name
        UNION ALL
        SELECT 'mob_drop' AS type, REPLACE(nt.name, '_', ' ') AS name, z.long_name AS zone
        FROM loottable_entries lte
        JOIN lootdrop_entries lde ON lde.lootdrop_id = lte.lootdrop_id
        JOIN npc_types nt ON nt.loottable_id = lte.loottable_id
        JOIN spawnentry se ON se.npcID = nt.id
        JOIN spawn2 s2 ON s2.spawngroupID = se.spawngroupID
        JOIN zone z ON z.short_name = s2.zone
        WHERE lde.item_id = p_item_id
        GROUP BY nt.name, z.long_name
    ) AS source_details;

    result := jsonb_build_object('type', 'obtainment', 'sources', obtainment_sources);
    RETURN result;
END;
$function$;
