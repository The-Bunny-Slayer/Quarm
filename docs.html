get_full_obtainment_details(p_item_id INT)

This database function takes an item ID and returns a JSON object representing the full, recursive crafting tree or drop information for that item.

SQL Definition


CREATE OR REPLACE FUNCTION get_full_obtainment_details(p_item_id INT)
RETURNS jsonb AS $$
DECLARE
    result jsonb;
    recipe_info RECORD;
    component_info RECORD;
    final_components jsonb[] := '{}';
    container_name text;
BEGIN
    -- First, find the primary recipe for the target item
    SELECT INTO recipe_info *
    FROM tradeskill_recipe r
    JOIN tradeskill_recipe_entries tre ON r.id = tre.recipe_id
    WHERE tre.item_id = p_item_id AND tre.successcount > 0
    LIMIT 1;

    -- If no recipe produces this item, find its drop source.
    IF NOT FOUND THEN
        SELECT jsonb_build_object(
            'type', 'drop',
            'sources', (
                SELECT jsonb_agg(source_details) FROM (
                    SELECT DISTINCT n.name AS npc_name, z.short_name AS zone_name
                    FROM loottable_entries lte
                    JOIN lootdrop_entries lde ON lde.lootdrop_id = lte.lootdrop_id
                    JOIN items i ON i.id = lde.item_id
                    JOIN npc_types n ON n.loottable_id = lte.loottable_id
                    JOIN spawnentry se ON se.npcID = n.id
                    JOIN spawn2 s ON s.spawngroupID = se.spawngroupID
                    JOIN zone z ON z.short_name = s.zone
                    WHERE i.id = p_item_id
                ) AS source_details
            )
        ) INTO result;
        RETURN result;
    END IF;

    -- If it is a crafted item, loop through its components
    FOR component_info IN
        SELECT i.id as item_id, i.name as item_name, tre.componentcount as qty
        FROM tradeskill_recipe_entries tre
        JOIN items i ON i.id = tre.item_id
        WHERE tre.recipe_id = recipe_info.recipe_id AND tre.componentcount > 0
    LOOP
        -- For each component, recursively call this function to get its breakdown
        final_components := array_append(final_components, jsonb_build_object(
            'item_id', component_info.item_id,
            'item_name', component_info.item_name,
            'qty', component_info.qty,
            'obtainment', get_full_obtainment_details(component_info.item_id)
        ));
    END LOOP;

    -- Get container info
    SELECT i.name INTO container_name
    FROM tradeskill_recipe_entries tre
    JOIN items i ON i.id = tre.item_id
    WHERE tre.recipe_id = recipe_info.recipe_id AND tre.iscontainer IS TRUE
    LIMIT 1;

    IF NOT FOUND THEN
        container_name := 'None';
    END IF;

    -- Assemble the final JSON for the crafted item
    result := jsonb_build_object(
        'type', 'craft',
        'recipe_name', recipe_info.name,
        'tradeskill', recipe_info.tradeskill,
        'trivial', recipe_info.trivial,
        'container', container_name,
        'components', final_components
    );

    RETURN result;
END;
$$ LANGUAGE plpgsql;
